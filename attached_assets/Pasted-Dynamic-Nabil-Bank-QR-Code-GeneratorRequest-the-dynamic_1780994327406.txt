escpos.USB = require('escpos-usb'); // Or escpos.Network for IP printers

const device  = new escpos.USB(); // Automatically discovers local USB printers
const printer = new escpos.Printer(device);

function printLHCReceipt(orderData) {
    device.open(function(error){
        if(error) return console.error(error);
        
        printer
        .font('a')
        .align('ct')
        .style('bu')
        .size(2, 2)
        .text('LHC CAFE')
        .size(1, 1)
        .text('Kathmandu, Nepal')
        .text('--------------------------------')
        .align('lt')
        .text(`Order ID: ${orderData.id}`)
        .text(`Date: ${new Date().toLocaleDateString()}`)
        .text('--------------------------------');
        
        orderData.items.forEach(item => {
            printer.text(`${item.name} x ${item.qty}   Rs. ${item.price}`);
        });
        
        printer
        .text('--------------------------------')
        .size(1.5, 1.5)
        .text(`TOTAL: Rs. ${orderData.total}`)
        .size(1, 1)
        .text('Thank you for visiting LHC!')
        .feed(3)
        .cut() // Drops paper slice
        .close();
    });
}
